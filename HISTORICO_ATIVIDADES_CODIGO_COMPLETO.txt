-- Append-only student activity history. Binary file contents and session secrets
-- are intentionally excluded from event payloads.
CREATE TABLE IF NOT EXISTS metalab_student_activity_events (
  id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  student_id UUID NOT NULL REFERENCES metalab_students(id) ON DELETE RESTRICT,
  actor_student_id UUID REFERENCES metalab_students(id) ON DELETE SET NULL,
  class_id UUID,
  occurred_at TIMESTAMPTZ NOT NULL,
  actor_role VARCHAR(12) NOT NULL CHECK (actor_role IN ('student','teacher','system')),
  event_type VARCHAR(80) NOT NULL,
  entity_type VARCHAR(40) NOT NULL,
  entity_id TEXT NOT NULL,
  payload JSONB NOT NULL DEFAULT '{}'::JSONB,
  source_table VARCHAR(80) NOT NULL,
  source_row_id TEXT NOT NULL,
  legacy BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT clock_timestamp()
);

CREATE INDEX IF NOT EXISTS metalab_student_activity_student_time_idx
  ON metalab_student_activity_events (student_id, occurred_at DESC, id DESC);
CREATE INDEX IF NOT EXISTS metalab_student_activity_actor_time_idx
  ON metalab_student_activity_events (actor_student_id, occurred_at DESC, id DESC)
  WHERE actor_student_id IS NOT NULL;
CREATE INDEX IF NOT EXISTS metalab_student_activity_class_time_idx
  ON metalab_student_activity_events (class_id, occurred_at DESC, id DESC)
  WHERE class_id IS NOT NULL;
CREATE UNIQUE INDEX IF NOT EXISTS metalab_student_activity_legacy_source_uidx
  ON metalab_student_activity_events (source_table, source_row_id, event_type)
  WHERE legacy = TRUE;

CREATE OR REPLACE FUNCTION metalab_record_student_activity(
  p_student_id UUID,
  p_event_type TEXT,
  p_entity_type TEXT,
  p_entity_id TEXT,
  p_payload JSONB,
  p_source_table TEXT,
  p_source_row_id TEXT,
  p_class_id UUID DEFAULT NULL,
  p_actor_role TEXT DEFAULT NULL,
  p_actor_student_id UUID DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT NULL,
  p_legacy BOOLEAN DEFAULT FALSE
) RETURNS VOID LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE
  v_role TEXT;
BEGIN
  IF p_student_id IS NULL THEN RETURN; END IF;
  v_role := COALESCE(
    NULLIF(current_setting('metalab.actor_role', TRUE), ''),
    p_actor_role,
    'system'
  );
  IF v_role NOT IN ('student','teacher','system') THEN v_role := 'system'; END IF;

  INSERT INTO metalab_student_activity_events (
    student_id, actor_student_id, class_id, occurred_at, actor_role,
    event_type, entity_type, entity_id, payload, source_table,
    source_row_id, legacy
  ) VALUES (
    p_student_id, p_actor_student_id, p_class_id,
    COALESCE(p_occurred_at, clock_timestamp()), v_role,
    p_event_type, p_entity_type, p_entity_id, COALESCE(p_payload, '{}'::JSONB),
    p_source_table, p_source_row_id, p_legacy
  ) ON CONFLICT (source_table, source_row_id, event_type) WHERE legacy = TRUE
    DO NOTHING;
END $$;

CREATE OR REPLACE FUNCTION metalab_capture_student_activity() RETURNS TRIGGER
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE
  v_new JSONB;
  v_old JSONB;
  v_student UUID;
  v_actor UUID;
  v_class UUID;
  v_post_owner UUID;
  v_post_body TEXT;
  v_post_image_name TEXT;
  v_post_image_mime TEXT;
  v_post_has_image BOOLEAN;
  v_event TEXT;
  v_payload JSONB;
  v_role TEXT;
  v_entity_id TEXT;
  v_class_name TEXT;
  v_count BIGINT;
  v_like_count BIGINT;
BEGIN
  IF TG_OP <> 'DELETE' THEN
    IF TG_TABLE_NAME='metalab_student_photos' THEN
      v_new:=jsonb_build_object('id',NEW.id,'student_id',NEW.student_id,'file_name',NEW.file_name,'product',NEW.product,'angle',NEW.angle,'height',NEW.height,'mime_type',NEW.mime_type,'size_bytes',NEW.size_bytes,'created_at',NEW.created_at);
    ELSIF TG_TABLE_NAME='metalab_student_deliveries' THEN
      v_new:=jsonb_build_object('id',NEW.id,'student_id',NEW.student_id,'activity_name',NEW.activity_name,'description',NEW.description,'class_date',NEW.class_date,'file_name',NEW.file_name,'mime_type',NEW.mime_type,'size_bytes',NEW.size_bytes,'created_at',NEW.created_at,'quick_mission_id',NEW.quick_mission_id);
    ELSIF TG_TABLE_NAME='metalab_instalab_posts' THEN
      v_new:=jsonb_build_object('id',NEW.id,'student_id',NEW.student_id,'class_id',NEW.class_id,'body',NEW.body,'image_name',NEW.image_name,'image_mime_type',NEW.image_mime_type,'image_size_bytes',NEW.image_size_bytes,'image_present',NEW.image_data IS NOT NULL,'created_at',NEW.created_at);
    ELSIF TG_TABLE_NAME='metalab_student_sessions' THEN
      v_new:=jsonb_build_object('id',NEW.id,'student_id',NEW.student_id,'created_at',NEW.created_at,'revoked_at',NEW.revoked_at);
    ELSE v_new:=to_jsonb(NEW);
    END IF;
  END IF;
  IF TG_OP <> 'INSERT' THEN
    IF TG_TABLE_NAME='metalab_student_photos' THEN
      v_old:=jsonb_build_object('id',OLD.id,'student_id',OLD.student_id,'file_name',OLD.file_name,'product',OLD.product,'angle',OLD.angle,'height',OLD.height,'mime_type',OLD.mime_type,'size_bytes',OLD.size_bytes,'created_at',OLD.created_at);
    ELSIF TG_TABLE_NAME='metalab_student_deliveries' THEN
      v_old:=jsonb_build_object('id',OLD.id,'student_id',OLD.student_id,'activity_name',OLD.activity_name,'description',OLD.description,'class_date',OLD.class_date,'file_name',OLD.file_name,'mime_type',OLD.mime_type,'size_bytes',OLD.size_bytes,'created_at',OLD.created_at,'quick_mission_id',OLD.quick_mission_id);
    ELSIF TG_TABLE_NAME='metalab_instalab_posts' THEN
      v_old:=jsonb_build_object('id',OLD.id,'student_id',OLD.student_id,'class_id',OLD.class_id,'body',OLD.body,'image_name',OLD.image_name,'image_mime_type',OLD.image_mime_type,'image_size_bytes',OLD.image_size_bytes,'image_present',OLD.image_data IS NOT NULL,'created_at',OLD.created_at);
    ELSIF TG_TABLE_NAME='metalab_student_sessions' THEN
      v_old:=jsonb_build_object('id',OLD.id,'student_id',OLD.student_id,'created_at',OLD.created_at,'revoked_at',OLD.revoked_at);
    ELSE v_old:=to_jsonb(OLD);
    END IF;
  END IF;
  v_student := COALESCE((v_new->>'student_id')::UUID, (v_old->>'student_id')::UUID);
  v_entity_id := COALESCE(v_new->>'id', v_old->>'id', v_new->>'mission_id', v_old->>'mission_id');
  v_role := NULLIF(current_setting('metalab.actor_role', TRUE), '');

  IF TG_TABLE_NAME = 'metalab_businesses' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID, (v_old->>'student_id')::UUID);
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;
    v_entity_id := COALESCE(v_new->>'id', v_old->>'id');
    IF TG_OP = 'INSERT' THEN
      v_event := 'business.created';
      v_payload := jsonb_build_object('after', jsonb_build_object(
        'name',v_new->'name','segment',v_new->'segment','description',v_new->'description',
        'mainOffering',v_new->'main_offering','targetAudience',v_new->'target_audience',
        'differentiator',v_new->'differentiator','primaryColor',v_new->'primary_color'));
    ELSIF TG_OP = 'UPDATE' AND ROW(v_old->'name',v_old->'segment',v_old->'description',v_old->'main_offering',v_old->'target_audience',v_old->'differentiator',v_old->'primary_color') IS DISTINCT FROM ROW(v_new->'name',v_new->'segment',v_new->'description',v_new->'main_offering',v_new->'target_audience',v_new->'differentiator',v_new->'primary_color') THEN
      v_event := 'business.updated';
      v_payload := jsonb_build_object(
        'before',jsonb_build_object('name',v_old->'name','segment',v_old->'segment','description',v_old->'description','mainOffering',v_old->'main_offering','targetAudience',v_old->'target_audience','differentiator',v_old->'differentiator','primaryColor',v_old->'primary_color'),
        'after',jsonb_build_object('name',v_new->'name','segment',v_new->'segment','description',v_new->'description','mainOffering',v_new->'main_offering','targetAudience',v_new->'target_audience','differentiator',v_new->'differentiator','primaryColor',v_new->'primary_color'));
    ELSE RETURN NULL; END IF;
    PERFORM metalab_record_student_activity(v_student,v_event,'business',v_entity_id,v_payload,TG_TABLE_NAME,v_entity_id,v_class,COALESCE(v_role,'student'));

  ELSIF TG_TABLE_NAME = 'metalab_mission_progress' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID, (v_old->>'student_id')::UUID);
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;
    v_entity_id := COALESCE(v_new->>'mission_id',v_old->>'mission_id');
    IF TG_OP = 'INSERT' THEN
      v_event := 'mission.started';
      v_payload := jsonb_build_object('status',v_new->'status','answer',v_new->'answer','evidence',v_new->'evidence','feedback',v_new->'feedback','attemptCount',v_new->'attempt_count');
      PERFORM metalab_record_student_activity(v_student,v_event,'mission',v_entity_id,v_payload,TG_TABLE_NAME,v_new->>'id',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'started_at')::TIMESTAMPTZ);
      IF v_new->>'status' = 'completed' THEN
        PERFORM metalab_record_student_activity(v_student,'mission.completed','mission',v_entity_id,v_payload,TG_TABLE_NAME,v_new->>'id'||':completed',v_class,COALESCE(v_role,'student'),NULL,COALESCE((v_new->>'completed_at')::TIMESTAMPTZ,clock_timestamp()));
      END IF;
      RETURN NEW;
    ELSIF TG_OP = 'UPDATE' AND ROW(v_old->'status',v_old->'answer',v_old->'evidence',v_old->'feedback',v_old->'attempt_count') IS DISTINCT FROM ROW(v_new->'status',v_new->'answer',v_new->'evidence',v_new->'feedback',v_new->'attempt_count') THEN
      v_event := CASE WHEN v_old->>'status' <> 'completed' AND v_new->>'status' = 'completed' THEN 'mission.completed' ELSE 'mission.attempted' END;
      v_payload := jsonb_build_object('before',jsonb_build_object('status',v_old->'status','answer',v_old->'answer','evidence',v_old->'evidence','feedback',v_old->'feedback','attemptCount',v_old->'attempt_count'),'after',jsonb_build_object('status',v_new->'status','answer',v_new->'answer','evidence',v_new->'evidence','feedback',v_new->'feedback','attemptCount',v_new->'attempt_count'));
      PERFORM metalab_record_student_activity(v_student,v_event,'mission',v_entity_id,v_payload,TG_TABLE_NAME,v_new->>'id'||':'||COALESCE(v_new->>'attempt_count','update'),v_class,COALESCE(v_role,'student'),NULL,clock_timestamp());
    ELSE RETURN NULL; END IF;

  ELSIF TG_TABLE_NAME = 'metalab_teacher_task_submissions' THEN
    v_student := (v_new->>'student_id')::UUID;
    SELECT assignment.class_id,classroom.name INTO v_class,v_class_name FROM metalab_student_class_assignments assignment JOIN metalab_classes classroom ON classroom.id=assignment.class_id WHERE assignment.student_id = v_student;
    SELECT jsonb_build_object('prompt',task.prompt,'missionTitle',mission.title,'moduleTitle',module.title) INTO v_payload
      FROM metalab_teacher_tasks task JOIN metalab_teacher_missions mission ON mission.id=task.mission_id
      JOIN metalab_teacher_modules module ON module.id=mission.module_id
      WHERE task.id=(v_new->>'task_id')::UUID;
    v_payload := COALESCE(v_payload,'{}'::JSONB) || jsonb_build_object('answer',v_new->'answer','taskId',v_new->'task_id','submittedAt',v_new->'submitted_at','responseId',v_new->'response_id','classId',v_class,'className',v_class_name);
    PERFORM metalab_record_student_activity(v_student,'classroom.task.submitted','classroom_task',v_new->>'task_id',v_payload,TG_TABLE_NAME,v_new->>'id',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'submitted_at')::TIMESTAMPTZ);

  ELSIF TG_TABLE_NAME = 'metalab_student_deliveries' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;
    IF TG_OP='INSERT' THEN
      v_event := CASE WHEN v_new->>'quick_mission_id' IS NULL THEN 'delivery.submitted' ELSE 'quick_mission.submitted' END;
      v_payload := jsonb_build_object('activityName',v_new->'activity_name','description',v_new->'description','classDate',v_new->'class_date','fileName',v_new->'file_name','mimeType',v_new->'mime_type','sizeBytes',v_new->'size_bytes','quickMissionId',v_new->'quick_mission_id');
      PERFORM metalab_record_student_activity(v_student,v_event,'delivery',v_new->>'id',v_payload,TG_TABLE_NAME,v_new->>'id',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'created_at')::TIMESTAMPTZ);
    ELSIF TG_OP='DELETE' THEN
      v_event := CASE WHEN v_old->>'quick_mission_id' IS NULL THEN 'delivery.deleted' ELSE 'quick_mission.delivery_deleted' END;
      v_payload := jsonb_build_object('activityName',v_old->'activity_name','description',v_old->'description','classDate',v_old->'class_date','fileName',v_old->'file_name','mimeType',v_old->'mime_type','quickMissionId',v_old->'quick_mission_id');
      PERFORM metalab_record_student_activity(v_student,v_event,'delivery',v_old->>'id',v_payload,TG_TABLE_NAME,v_old->>'id'||':deleted:'||clock_timestamp(),v_class,COALESCE(v_role,'student'),NULL,clock_timestamp());
    ELSE RETURN NULL; END IF;

  ELSIF TG_TABLE_NAME = 'metalab_student_photos' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;
    IF TG_OP='INSERT' THEN
      v_payload := jsonb_build_object('fileName',v_new->'file_name','product',v_new->'product','angle',v_new->'angle','height',v_new->'height','mimeType',v_new->'mime_type','sizeBytes',v_new->'size_bytes','receivedAt',v_new->'created_at','captureTimeTrusted',false);
      PERFORM metalab_record_student_activity(v_student,'camera.photo_received','photo',v_new->>'id',v_payload,TG_TABLE_NAME,v_new->>'id',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'created_at')::TIMESTAMPTZ);
    ELSIF TG_OP='DELETE' THEN
      v_payload := jsonb_build_object('fileName',v_old->'file_name','product',v_old->'product','angle',v_old->'angle','height',v_old->'height','mimeType',v_old->'mime_type','sizeBytes',v_old->'size_bytes','removed',true);
      PERFORM metalab_record_student_activity(v_student,'camera.photo_removed','photo',v_old->>'id',v_payload,TG_TABLE_NAME,v_old->>'id'||':deleted:'||clock_timestamp(),v_class,COALESCE(v_role,'student'),NULL,clock_timestamp());
    ELSIF TG_OP='UPDATE' AND (ROW(v_old->'file_name',v_old->'product',v_old->'angle',v_old->'height',v_old->'mime_type',v_old->'size_bytes') IS DISTINCT FROM ROW(v_new->'file_name',v_new->'product',v_new->'angle',v_new->'height',v_new->'mime_type',v_new->'size_bytes') OR OLD.image_data IS DISTINCT FROM NEW.image_data) THEN
      v_payload:=jsonb_build_object('before',jsonb_build_object('fileName',v_old->'file_name','product',v_old->'product','angle',v_old->'angle','height',v_old->'height','mimeType',v_old->'mime_type','sizeBytes',v_old->'size_bytes'),'after',jsonb_build_object('fileName',v_new->'file_name','product',v_new->'product','angle',v_new->'angle','height',v_new->'height','mimeType',v_new->'mime_type','sizeBytes',v_new->'size_bytes'));
      PERFORM metalab_record_student_activity(v_student,'camera.photo_updated','photo',v_new->>'id',v_payload,TG_TABLE_NAME,v_new->>'id'||':update:'||clock_timestamp(),v_class,COALESCE(v_role,'student'));
    ELSE RETURN NULL; END IF;

  ELSIF TG_TABLE_NAME = 'metalab_instalab_posts' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    v_class := COALESCE((v_new->>'class_id')::UUID,(v_old->>'class_id')::UUID);
    IF TG_OP='INSERT' THEN
      v_event := 'instalab.post_created';
      v_payload := jsonb_build_object('body',v_new->'body','imageName',v_new->'image_name','imageMimeType',v_new->'image_mime_type','hasImage',v_new->'image_present');
      PERFORM metalab_record_student_activity(v_student,v_event,'instalab_post',v_new->>'id',v_payload,TG_TABLE_NAME,v_new->>'id',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'created_at')::TIMESTAMPTZ);
    ELSIF TG_OP='UPDATE' AND ROW(v_old->'body',v_old->'image_name',v_old->'image_mime_type',v_old->'image_size_bytes',v_old->'image_present') IS DISTINCT FROM ROW(v_new->'body',v_new->'image_name',v_new->'image_mime_type',v_new->'image_size_bytes',v_new->'image_present') THEN
      v_payload := jsonb_build_object('before',jsonb_build_object('body',v_old->'body','imageName',v_old->'image_name','imageMimeType',v_old->'image_mime_type','hasImage',v_old->'image_present'),'after',jsonb_build_object('body',v_new->'body','imageName',v_new->'image_name','imageMimeType',v_new->'image_mime_type','hasImage',v_new->'image_present'));
      PERFORM metalab_record_student_activity(v_student,'instalab.post_edited','instalab_post',v_new->>'id',v_payload,TG_TABLE_NAME,v_new->>'id'||':edit:'||clock_timestamp(),v_class,COALESCE(v_role,'student'));
    ELSIF TG_OP='DELETE' THEN
      SELECT COUNT(*) INTO v_count FROM metalab_instalab_comments WHERE post_id=(v_old->>'id')::UUID;
      SELECT COUNT(*) INTO v_like_count FROM metalab_instalab_likes WHERE post_id=(v_old->>'id')::UUID;
      v_payload := jsonb_build_object('body',v_old->'body','imageName',v_old->'image_name','imageMimeType',v_old->'image_mime_type','hasImage',v_old->'image_present','commentsPresentAtRemoval',v_count,'likesPresentAtRemoval',v_like_count);
      PERFORM metalab_record_student_activity(v_student,'instalab.post_deleted','instalab_post',v_old->>'id',v_payload,TG_TABLE_NAME,v_old->>'id'||':deleted:'||clock_timestamp(),v_class,COALESCE(v_role,'teacher'));
    ELSE RETURN NULL; END IF;

  ELSIF TG_TABLE_NAME = 'metalab_instalab_likes' THEN
    v_class := NULL;
    SELECT post.student_id,post.class_id,post.body,post.image_name,post.image_mime_type,(post.image_data IS NOT NULL) INTO v_post_owner,v_class,v_post_body,v_post_image_name,v_post_image_mime,v_post_has_image FROM metalab_instalab_posts post WHERE post.id=COALESCE((v_new->>'post_id')::UUID,(v_old->>'post_id')::UUID);
    IF v_post_owner IS NULL THEN RETURN NULL; END IF;
    v_student := v_post_owner;
    v_actor := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    v_role := CASE WHEN COALESCE((v_new->>'teacher_like')::BOOLEAN,(v_old->>'teacher_like')::BOOLEAN,FALSE) THEN 'teacher' ELSE COALESCE(v_role,'student') END;
    v_event := CASE WHEN TG_OP='INSERT' THEN 'instalab.like_added' ELSE 'instalab.like_removed' END;
    v_payload := jsonb_build_object('postId',COALESCE(v_new->'post_id',v_old->'post_id'),'actorName',(SELECT student.full_name FROM metalab_students student WHERE student.id=v_actor),'teacherLike',COALESCE((v_new->>'teacher_like')::BOOLEAN,(v_old->>'teacher_like')::BOOLEAN,FALSE),'postSnapshot',jsonb_build_object('body',v_post_body,'imageName',v_post_image_name,'imageMimeType',v_post_image_mime,'hasImage',v_post_has_image));
    PERFORM metalab_record_student_activity(v_student,v_event,'instalab_post',COALESCE(v_new->>'post_id',v_old->>'post_id'),v_payload,TG_TABLE_NAME,COALESCE(v_new->>'post_id',v_old->>'post_id')||':'||COALESCE(v_new->>'student_id','teacher')||':'||clock_timestamp(),v_class,v_role,v_actor);

  ELSIF TG_TABLE_NAME = 'metalab_instalab_comments' THEN
    SELECT post.student_id,post.class_id,post.body,post.image_name,post.image_mime_type,(post.image_data IS NOT NULL) INTO v_post_owner,v_class,v_post_body,v_post_image_name,v_post_image_mime,v_post_has_image FROM metalab_instalab_posts post WHERE post.id=COALESCE((v_new->>'post_id')::UUID,(v_old->>'post_id')::UUID);
    IF v_post_owner IS NULL THEN RETURN NULL; END IF;
    v_student := v_post_owner;
    v_actor := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    v_role := COALESCE(v_role,v_new->>'author_role',v_old->>'author_role','student');
    v_event := CASE WHEN TG_OP='INSERT' THEN 'instalab.comment_added' ELSE 'instalab.comment_removed' END;
    v_payload := jsonb_build_object('postId',COALESCE(v_new->'post_id',v_old->'post_id'),'commentId',COALESCE(v_new->'id',v_old->'id'),'body',COALESCE(v_new->'body',v_old->'body'),'authorName',COALESCE(v_new->'author_name',v_old->'author_name'),'postSnapshot',jsonb_build_object('body',v_post_body,'imageName',v_post_image_name,'imageMimeType',v_post_image_mime,'hasImage',v_post_has_image));
    PERFORM metalab_record_student_activity(v_student,v_event,'instalab_post',COALESCE(v_new->>'post_id',v_old->>'post_id'),v_payload,TG_TABLE_NAME,COALESCE(v_new->>'id',v_old->>'id')||':'||TG_OP||':'||clock_timestamp(),v_class,v_role,v_actor);

  ELSIF TG_TABLE_NAME = 'metalab_student_class_assignments' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    SELECT name INTO v_class_name FROM metalab_classes WHERE id=(v_new->>'class_id')::UUID;
    IF TG_OP='INSERT' THEN
      v_class := (v_new->>'class_id')::UUID;
      v_event := 'class.joined';
      v_payload := jsonb_build_object('className',v_class_name,'assignedAt',v_new->'assigned_at');
    ELSIF TG_OP='DELETE' THEN
      v_class := (v_old->>'class_id')::UUID;
      SELECT name INTO v_class_name FROM metalab_classes WHERE id=v_class;
      v_event := 'class.left';
      v_payload := jsonb_build_object('className',v_class_name,'removed',true);
    ELSE
      v_class := (v_new->>'class_id')::UUID;
      SELECT name INTO v_class_name FROM metalab_classes WHERE id=(v_old->>'class_id')::UUID;
      v_event := 'class.transferred';
      v_payload := jsonb_build_object('fromClassId',v_old->'class_id','fromClassName',v_class_name,'toClassId',v_new->'class_id','toClassName',(SELECT name FROM metalab_classes WHERE id=v_class),'assignedAt',v_new->'assigned_at');
    END IF;

  ELSIF TG_TABLE_NAME = 'metalab_student_sessions' THEN
    v_student := COALESCE((v_new->>'student_id')::UUID,(v_old->>'student_id')::UUID);
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;

    ELSIF TG_OP='UPDATE' AND v_old->>'revoked_at' IS NULL AND v_new->>'revoked_at' IS NOT NULL THEN
      PERFORM metalab_record_student_activity(v_student,'session.ended','session',v_new->>'id',jsonb_build_object('endedAt',v_new->'revoked_at'),TG_TABLE_NAME,v_new->>'id'||':ended',v_class,COALESCE(v_role,'student'),NULL,(v_new->>'revoked_at')::TIMESTAMPTZ);
    ELSE RETURN NULL; END IF;

  ELSIF TG_TABLE_NAME = 'metalab_students' THEN
    v_student := (v_new->>'id')::UUID;
    SELECT class_id INTO v_class FROM metalab_student_class_assignments WHERE student_id=v_student;
    IF TG_OP='INSERT' THEN
      PERFORM metalab_record_student_activity(v_student,'student.registered','student',v_new->>'id',jsonb_build_object('fullName',v_new->'full_name','createdAt',v_new->'created_at'),TG_TABLE_NAME,v_new->>'id',NULL,'student',NULL,(v_new->>'created_at')::TIMESTAMPTZ);
    ELSIF TG_OP='UPDATE' AND v_old->>'archived_at' IS DISTINCT FROM v_new->>'archived_at' THEN
      v_event := CASE WHEN v_new->>'archived_at' IS NULL THEN 'student.restored' ELSE 'student.archived' END;
      PERFORM metalab_record_student_activity(v_student,v_event,'student',v_new->>'id',jsonb_build_object('archivedAt',v_new->'archived_at'),TG_TABLE_NAME,v_new->>'id'||':'||v_event||':'||clock_timestamp(),v_class,COALESCE(v_role,'teacher'));
    ELSE RETURN NULL; END IF;
  ELSE
    RETURN NULL;
  END IF;

  IF TG_OP='DELETE' THEN RETURN OLD; END IF;
  RETURN NEW;
END $$;

import { NextRequest, NextResponse } from 'next/server';
import { getTeacherStudent } from '@/server/repository';
import { getPool } from '@/server/db';
import { hasTeacherSession } from '@/server/teacher-auth';
import { z } from 'zod';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

type RouteContext = { params: Promise<{ studentId: string }> };
const isoDate = z.string().regex(/^\d{4}-\d{2}-\d{2}$/u).refine(value => {
  const date = new Date(`${value}T00:00:00.000Z`);
  return !Number.isNaN(date.getTime()) && date.toISOString().slice(0, 10) === value;
});
const querySchema = z.object({
  from: isoDate.optional(),
  to: isoDate.optional(),
  limit: z.coerce.number().int().min(1).max(100).default(50),
  offset: z.coerce.number().int().min(0).max(100000).default(0)
}).strict();

export async function GET(request: NextRequest, context: RouteContext) {
  if (!hasTeacherSession(request)) return NextResponse.json({ error: 'Sessão do professor necessária.' }, { status: 401, headers: { 'Cache-Control': 'no-store' } });
  const studentId = (await context.params).studentId;
  if (!z.string().uuid().safeParse(studentId).success) return NextResponse.json({ error: 'Aluno não encontrado.' }, { status: 404 });
  const query = querySchema.safeParse(Object.fromEntries(request.nextUrl.searchParams));
  if (!query.success || (query.data.from && query.data.to && query.data.from > query.data.to)) {
    return NextResponse.json({ error: 'Período inválido.' }, { status: 400, headers: { 'Cache-Control': 'no-store' } });
  }

  try {
    // Keep the same student authorization boundary as the profile endpoint.
    const student = await getTeacherStudent(studentId);
    if (!student) return NextResponse.json({ error: 'Aluno não encontrado.' }, { status: 404 });
    const { from = null, to = null, limit, offset } = query.data;
    const pool = getPool();
    const dateFilter = `($2::date IS NULL OR event.occurred_at >= ($2::date::timestamp AT TIME ZONE 'America/Sao_Paulo'))
      AND ($3::date IS NULL OR event.occurred_at < (($3::date + 1)::timestamp AT TIME ZONE 'America/Sao_Paulo'))`;
    const [events, count, days] = await Promise.all([
      pool.query(`SELECT event.id::text, event.actor_student_id, event.class_id,
          event.occurred_at, event.actor_role, event.event_type, event.entity_type,
          event.entity_id, event.payload, event.legacy
        FROM metalab_student_activity_events event
        WHERE (event.student_id = $1 OR event.actor_student_id = $1) AND ${dateFilter}
        ORDER BY event.occurred_at ASC, event.id ASC
        LIMIT $4 OFFSET $5`, [studentId, from, to, limit, offset]),
      pool.query(`SELECT COUNT(*)::int AS total FROM metalab_student_activity_events event
        WHERE (event.student_id = $1 OR event.actor_student_id = $1) AND ${dateFilter}`, [studentId, from, to]),
      pool.query(`SELECT DISTINCT (event.occurred_at AT TIME ZONE 'America/Sao_Paulo')::date::text AS day
        FROM metalab_student_activity_events event
        WHERE (event.student_id = $1 OR event.actor_student_id = $1)
        ORDER BY day DESC LIMIT 180`, [studentId])
    ]);
    return NextResponse.json({
      events: events.rows.map(row => ({
        id: row.id,
        occurredAt: new Date(row.occurred_at).toISOString(),
        actorRole: row.actor_role,
        eventType: row.event_type,
        entityType: row.entity_type,
        entityId: row.entity_id,
        classId: row.class_id,
        payload: row.payload,
        legacy: row.legacy === true,
        actorIsStudent: row.actor_student_id !== null
      })),
      total: count.rows[0]?.total ?? 0,
      limit,
      offset,
      hasMore: offset + events.rowCount! < Number(count.rows[0]?.total ?? 0),
      recordedDays: days.rows.map(row => row.day)
    }, { headers: { 'Cache-Control': 'no-store' } });
  } catch (error) {
    const code = error && typeof error === 'object' && 'code' in error ? String(error.code) : 'UNEXPECTED';
    console.error(`teacher.student.history.read failed (${code})`);
    const message = code === '42P01'
      ? 'O histórico ainda não está instalado neste banco. Depois de conferir o destino e ter um backup, aplique as migrations pendentes.'
      : 'Não foi possível carregar o histórico deste aluno agora.';
    return NextResponse.json({ error: message }, { status: 503, headers: { 'Cache-Control': 'no-store' } });
  }
}

export async function createStudentDelivery(input: {
  studentId: string; activityName: string; description: string; classDate: string; quickMissionId?: string | null;
  fileName: string; mimeType: string; fileData: Buffer;
}) {
  const result = await getPool().query<DeliveryRow>(`
    INSERT INTO metalab_student_deliveries
      (id, student_id, activity_name, description, class_date, file_name, mime_type, size_bytes, file_data, quick_mission_id)
    SELECT $1,$2,COALESCE(mission.title,$3),$4,$5,$6,$7,$8,$9,$10
    FROM (SELECT 1) seed
    LEFT JOIN metalab_quick_missions mission ON mission.id = $10::uuid
    LEFT JOIN metalab_quick_modules module ON module.id=mission.module_id AND module.is_published=TRUE
    WHERE $10::uuid IS NULL OR (mission.id IS NOT NULL AND module.id IS NOT NULL)
    RETURNING id, activity_name, description, class_date, file_name, mime_type, size_bytes, created_at, quick_mission_id
  `, [randomUUID(), input.studentId, input.activityName, input.description, input.classDate,
    input.fileName, input.mimeType, input.fileData.byteLength, input.fileData, input.quickMissionId ?? null]);
  if (!result.rows[0]) throw new Error('QUICK_MISSION_UNAVAILABLE');
  return mapDelivery(result.rows[0]);
}

export async function listStudentDeliveries(studentId: string) {
  const result = await getPool().query<DeliveryRow>(`
    SELECT delivery.id, delivery.activity_name, delivery.description, delivery.class_date, delivery.file_name, delivery.mime_type, delivery.size_bytes, delivery.created_at,
           delivery.quick_mission_id, mission.title AS quick_mission_title
    FROM metalab_student_deliveries delivery
    LEFT JOIN metalab_quick_missions mission ON mission.id = delivery.quick_mission_id
    WHERE delivery.student_id = $1 ORDER BY delivery.created_at DESC
  `, [studentId]);
  return result.rows.map(mapDelivery);
}

export async function listQuickMissionsForStudent(studentId: string): Promise<QuickModule[]> {
  const result = await getPool().query<any>(`
    SELECT module.id module_id,module.title module_title,module.description module_description,module.position module_position,
      mission.id,mission.title,mission.description,mission.request,mission.estimated_minutes,mission.due_at,
      mission.is_active,mission.created_at,
      (mission.due_at IS NOT NULL AND mission.due_at <= NOW()) AS is_expired,
      EXISTS (SELECT 1 FROM metalab_student_deliveries delivery WHERE delivery.quick_mission_id = mission.id AND delivery.student_id = $1) AS submitted
    FROM metalab_quick_modules module
    JOIN metalab_quick_missions mission ON mission.module_id=module.id
    WHERE module.is_published = TRUE
    ORDER BY module.position,mission.created_at DESC
  `, [studentId]);
  const modules = new Map<string, QuickModule>();
  for (const row of result.rows) {
    let module = modules.get(row.module_id);
    if (!module) {
      module = { id: row.module_id, title: row.module_title, description: row.module_description, position: row.module_position, isPublished: true, missions: [] };
      modules.set(module.id, module);
    }
    module.missions.push({ ...mapQuickMission(row), isExpired: row.is_expired, submitted: row.submitted });
  }
  return [...modules.values()];
}

    FROM metalab_quick_modules module LEFT JOIN metalab_quick_missions mission ON mission.module_id=module.id
    ORDER BY module.position,module.created_at,mission.created_at DESC
  `);
  const modules = new Map<string, QuickModule>();
  for (const row of result.rows) {
    let module = modules.get(row.module_id);
    if (!module) {
      module = { id: row.module_id, title: row.module_title, description: row.module_description, position: row.module_position, isPublished: row.is_published, missions: [] };
      modules.set(module.id, module);
    }
    if (row.id) module.missions.push(mapQuickMission(row));
  }
  return [...modules.values()];
}

export async function listClassroomForTeacher(): Promise<ClassroomModule[]> {
  const pool = getPool();
  const [modules, missions, tasks, submissions] = await Promise.all([
    pool.query<any>('SELECT id,title,description,position,is_published FROM metalab_teacher_modules ORDER BY position,created_at'),
    pool.query<any>('SELECT id,module_id,title,objective,description,position,is_published FROM metalab_teacher_missions ORDER BY position,created_at'),
    pool.query<any>(`SELECT id,mission_id,prompt,description,task_type,options,settings,is_required,position,is_published FROM metalab_teacher_tasks ORDER BY position,created_at`),
    pool.query<any>(`SELECT answer.task_id, COUNT(answer.id)::int AS count FROM metalab_teacher_task_submissions answer GROUP BY answer.task_id`)
  ]);
  const counts = new Map(submissions.rows.map((row: any) => [row.task_id, row.count]));
  const taskMap = new Map<string, ClassroomTask[]>();
  for (const row of tasks.rows) {
    const list = taskMap.get(row.mission_id) || [];
    list.push({ id: row.id, prompt: row.prompt, description: row.description || '', type: row.task_type, required: row.is_required !== false, settings: row.settings || {}, options: row.options || [], position: row.position, isPublished: row.is_published, submissionCount: Number(counts.get(row.id) || 0) });
    taskMap.set(row.mission_id, list);
  }
  const missionMap = new Map<string, ClassroomMission[]>();
  for (const row of missions.rows) {
    const list = missionMap.get(row.module_id) || [];
    list.push({ id: row.id, moduleId: row.module_id, title: row.title, objective: row.objective, description: row.description, position: row.position, isPublished: row.is_published, tasks: taskMap.get(row.id) || [] });
    missionMap.set(row.module_id, list);
  }
  return modules.rows.map((row: any) => ({ id: row.id, title: row.title, description: row.description, position: row.position, isPublished: row.is_published, missions: missionMap.get(row.id) || [] }));
}

export async function listClassroomForStudent(studentId: string): Promise<ClassroomModule[]> {
  const result = await getPool().query<any>(`
    SELECT module.id module_id,module.title module_title,module.description module_description,module.position module_position,
      mission.id mission_id,mission.title mission_title,mission.objective mission_objective,mission.description mission_description,mission.position mission_position,
      task.id task_id,task.prompt task_prompt,task.description task_description,task.task_type,task.options,task.settings task_settings,task.is_required task_required,task.position task_position,
      submission.id submission_id,submission.answer,submission.submitted_at
    FROM metalab_teacher_modules module
    LEFT JOIN metalab_teacher_missions mission ON mission.module_id=module.id
    LEFT JOIN metalab_teacher_tasks task ON task.mission_id=mission.id
    LEFT JOIN LATERAL (SELECT id,answer,submitted_at FROM metalab_teacher_task_submissions WHERE task_id=task.id AND student_id=$1 ORDER BY submitted_at DESC LIMIT 1) submission ON TRUE
    WHERE module.is_published=TRUE ORDER BY module.position,mission.position,task.position
  `, [studentId]);
  const modules = new Map<string, ClassroomModule>();
  for (const row of result.rows) {
    let module = modules.get(row.module_id);
    if (!module) { module = { id: row.module_id, title: row.module_title, description: row.module_description, position: row.module_position, isPublished: true, missions: [] }; modules.set(row.module_id, module); }
    if (row.mission_id) {
      let mission = module.missions.find(item => item.id === row.mission_id);
      if (!mission) { mission = { id: row.mission_id, moduleId: row.module_id, title: row.mission_title, objective: row.mission_objective, description: row.mission_description, position: row.mission_position, isPublished: true, tasks: [] }; module.missions.push(mission); }
      if (row.task_id) mission.tasks.push({ id: row.task_id, prompt: row.task_prompt, description: row.task_description || '', type: row.task_type, required: row.task_required !== false, settings: row.task_settings || {}, options: row.options || [], position: row.task_position, isPublished: true, ...(row.submission_id ? { submissions: [{ id: row.submission_id, studentId, answer: row.answer, submittedAt: toIso(row.submitted_at) }] } : {}) });
    }
  }
  return [...modules.values()];
}

export async function submitClassroomMissionResponse(input:{studentId:string;missionId:string;answers:Array<{taskId:string;answer:unknown}>}) {

export async function createOrResumeStudentWithSession(input: {
  name: string;
  email: string;
  birthDate: string;
  privacyNoticeVersion: string;
  photoSharingConsent: boolean;
}) {
  const session = createSessionMaterial();

  const result = await transaction(async client => {
    // Serialize retries for the same normalized email and birth date, preventing duplicate records.
    await client.query('SELECT pg_advisory_xact_lock(hashtextextended($1, 0))', [`${input.email}:${input.birthDate}`]);
    const match = await client.query<StudentRow>(`
      SELECT student.id, student.full_name, student.created_at, student.photo_sharing_consent
      FROM metalab_students student
      WHERE LOWER(student.email) = $1 AND student.birth_date = $2::date
        AND student.archived_at IS NULL
      ORDER BY
        (SELECT COUNT(*) FROM metalab_mission_progress progress
         WHERE progress.student_id = student.id AND progress.status = 'completed') DESC,
        (SELECT MAX(progress.updated_at) FROM metalab_mission_progress progress
         WHERE progress.student_id = student.id) DESC NULLS LAST,
        student.created_at ASC
      LIMIT 1
    `, [input.email, input.birthDate]);

    let student: StudentSummary;
    let resumed = Boolean(match.rows[0]);
    if (match.rows[0]) {
      const updated = await client.query<StudentRow>(`
        UPDATE metalab_students SET photo_sharing_consent = TRUE,
          photo_sharing_consent_at = COALESCE(photo_sharing_consent_at, NOW()), updated_at = NOW()
        WHERE id = $1
        RETURNING id, full_name, created_at, photo_sharing_consent
      `, [match.rows[0].id]);
      student = mapStudent(updated.rows[0]);
    } else {
      const archived = await client.query(`
        SELECT 1 FROM metalab_students
        WHERE LOWER(email) = $1 AND birth_date = $2::date AND archived_at IS NOT NULL
        LIMIT 1
      `, [input.email, input.birthDate]);
      if (archived.rowCount) throw new Error('STUDENT_ARCHIVED');

      const studentId = randomUUID();
      const created = await client.query<StudentRow>(`
        INSERT INTO metalab_students
          (id, full_name, email, birth_date, privacy_notice_version, photo_sharing_consent, photo_sharing_consent_at, updated_at)
        VALUES ($1, $2, $3, $4, $5, $6, CASE WHEN $6 THEN NOW() ELSE NULL END, NOW())
        RETURNING id, full_name, created_at, photo_sharing_consent
      `, [studentId, input.name, input.email, input.birthDate, input.privacyNoticeVersion, input.photoSharingConsent]);
      student = mapStudent(created.rows[0]);
      resumed = false;
    }

    await client.query(`
      INSERT INTO metalab_student_sessions
        (id, student_id, token_hash, expires_at)
      VALUES ($1, $2, $3, $4)
    `, [session.id, student.id, session.tokenHash, session.expiresAt]);

    return { student, resumed };
  });

  return { ...result, token: session.token };
}

export async function findStudentBySessionToken(token: string) {
  if (!isSessionToken(token)) return null;
  const tokenHash = hashSessionToken(token);
  const result = await getPool().query<StudentRow>(`
    WITH active_session AS (
      UPDATE metalab_student_sessions
      SET last_seen_at = NOW()
      WHERE token_hash = $1
        AND revoked_at IS NULL
        AND expires_at > NOW()
      RETURNING student_id
    )
    SELECT student.id, student.full_name, student.created_at, student.photo_sharing_consent
    FROM active_session session
    JOIN metalab_students student ON student.id = session.student_id
    WHERE student.archived_at IS NULL
  `, [tokenHash]);
  return result.rows[0] ? mapStudent(result.rows[0]) : null;
}

}

export async function listTeacherStudents(totalMissions: number, archived = false): Promise<TeacherStudentSummary[]> {
  const result = await getPool().query<{
    id: string;
    full_name: string;
    email: string;
    birth_date: Date | string;
    created_at: Date | string;
    business_name: string | null;
    business_segment: string | null;
    business_description: string | null;
    class_id: string | null;
    class_name: string | null;
    main_offering: string | null;
    target_audience: string | null;
    differentiator: string | null;
    classroom_completed: number | string;
    classroom_total: number | string;
    completed_missions: number | string;
    photo_count: number | string;
    last_activity: Date | string;
  }>(`
    SELECT student.id, student.full_name, student.email, student.birth_date, student.created_at,
           business.name AS business_name, business.segment AS business_segment,
           business.description AS business_description, business.main_offering, business.target_audience, business.differentiator,
           class_row.id AS class_id, class_row.name AS class_name,
           COUNT(DISTINCT progress.mission_id) FILTER (WHERE progress.status = 'completed') AS completed_missions,
           COUNT(DISTINCT photo.id) AS photo_count,
           (SELECT COUNT(DISTINCT submission.task_id) FROM metalab_teacher_task_submissions submission JOIN metalab_teacher_tasks task ON task.id=submission.task_id JOIN metalab_teacher_missions mission ON mission.id=task.mission_id JOIN metalab_teacher_modules module ON module.id=mission.module_id WHERE submission.student_id=student.id AND module.is_published) AS classroom_completed,
           (SELECT COUNT(*) FROM metalab_teacher_tasks task JOIN metalab_teacher_missions mission ON mission.id=task.mission_id JOIN metalab_teacher_modules module ON module.id=mission.module_id WHERE module.is_published) AS classroom_total,
           GREATEST(student.created_at, COALESCE(MAX(progress.updated_at), student.created_at),
             COALESCE(MAX(photo.created_at), student.created_at),
             COALESCE((SELECT MAX(delivery.created_at) FROM metalab_student_deliveries delivery WHERE delivery.student_id = student.id), student.created_at),
             COALESCE((SELECT MAX(answer.submitted_at) FROM metalab_teacher_task_submissions answer WHERE answer.student_id = student.id), student.created_at)) AS last_activity
    FROM metalab_students student
    LEFT JOIN metalab_businesses business ON business.student_id = student.id
    LEFT JOIN metalab_student_class_assignments assignment ON assignment.student_id=student.id
    LEFT JOIN metalab_classes class_row ON class_row.id=assignment.class_id
    LEFT JOIN metalab_mission_progress progress ON progress.student_id = student.id
    LEFT JOIN metalab_student_photos photo ON photo.student_id = student.id
    WHERE student.archived_at IS ${archived ? 'NOT ' : ''}NULL
    GROUP BY student.id, business.name, business.segment, business.description, business.main_offering, business.target_audience, business.differentiator, class_row.id, class_row.name
    ORDER BY last_activity DESC, student.full_name ASC
  `);
  return result.rows.map(row => ({
    id: row.id,
    fullName: row.full_name,
    email: row.email,
    birthDate: row.birth_date instanceof Date ? row.birth_date.toISOString().slice(0, 10) : String(row.birth_date).slice(0, 10),
    createdAt: toIso(row.created_at),
    businessName: row.business_name,
    businessSegment: row.business_segment,
    businessDescription: row.business_description,
    mainOffering: row.main_offering,
    targetAudience: row.target_audience,
    differentiator: row.differentiator,
    completedMissions: Number(row.completed_missions) + Number(row.classroom_completed),
    totalMissions: totalMissions + Number(row.classroom_total),
    photoCount: Number(row.photo_count),
    lastActivity: toIso(row.last_activity)
    ,classId: row.class_id,
    className: row.class_name
  }));
}

async function listInstalabPosts(studentId: string, authorOnly = false, teacherView = false): Promise<InstalabPost[]> {
  const result = await getPool().query<{
    id: string; student_id: string; class_id: string; author_name: string; business_name: string; class_name: string; body: string;
    image_name: string | null; image_mime_type: string | null; image_size_bytes: number | null;
    created_at: Date | string; like_count: number | string; liked_by_me: boolean; comments: InstalabComment[] | null;
  }>(`
    SELECT post.id, post.student_id, post.class_id, student.full_name AS author_name, business.name AS business_name, class.name AS class_name,
      post.body, post.image_name, post.image_mime_type, post.image_size_bytes, post.created_at,
      (SELECT COUNT(*) FROM metalab_instalab_likes likes WHERE likes.post_id = post.id) AS like_count,
      CASE WHEN $3::boolean
        THEN EXISTS (SELECT 1 FROM metalab_instalab_likes mine WHERE mine.post_id = post.id AND mine.teacher_like = TRUE)
        ELSE EXISTS (SELECT 1 FROM metalab_instalab_likes mine WHERE mine.post_id = post.id AND mine.student_id = $1)
      END AS liked_by_me,
      COALESCE((
        SELECT json_agg(comment_row ORDER BY comment_row."createdAt" ASC)
        FROM (
          SELECT comment.id, comment.student_id AS "studentId", COALESCE(comment.author_name, commenter.full_name) AS "authorName",
            comment.author_role AS "authorRole", comment.body, comment.created_at AS "createdAt"
          FROM metalab_instalab_comments comment
          LEFT JOIN metalab_students commenter ON commenter.id = comment.student_id
          WHERE comment.post_id = post.id ORDER BY comment.created_at DESC LIMIT 30
        ) comment_row
      ), '[]'::json) AS comments
    FROM metalab_instalab_posts post
    JOIN metalab_students student ON student.id = post.student_id
      AND ($3::boolean = TRUE OR student.archived_at IS NULL)
    JOIN metalab_businesses business ON business.id = post.business_id
    JOIN metalab_classes class ON class.id = post.class_id
    WHERE ($3::boolean = TRUE OR post.class_id = (SELECT assignment.class_id FROM metalab_student_class_assignments assignment WHERE assignment.student_id = $1))
      AND ($2::boolean = FALSE OR post.student_id = $1)
    ORDER BY post.created_at DESC LIMIT CASE WHEN $3::boolean THEN NULL::integer ELSE 100 END
  `, [studentId, authorOnly, teacherView]);
  return result.rows.map(row => ({
    id: row.id, studentId: row.student_id, classId: row.class_id, authorName: row.author_name, businessName: row.business_name, className: row.class_name,
    body: row.body, imageName: row.image_name, imageMimeType: row.image_mime_type,
    imageSizeBytes: row.image_size_bytes === null ? null : Number(row.image_size_bytes),
    createdAt: toIso(row.created_at), likeCount: Number(row.like_count), likedByMe: row.liked_by_me,
    isMine: row.student_id === studentId, comments: row.comments ?? []
  }));
}

export async function setInstalabLike(postId: string, studentId: string, liked: boolean) {
  if (liked) {
    await getPool().query(`
      INSERT INTO metalab_instalab_likes (post_id, student_id)
      SELECT post.id, $2 FROM metalab_instalab_posts post
      JOIN metalab_student_class_assignments assignment ON assignment.class_id = post.class_id AND assignment.student_id = $2
      JOIN metalab_businesses business ON business.student_id = $2
      WHERE post.id = $1
      ON CONFLICT (post_id, student_id) DO NOTHING
    `, [postId, studentId]);
  } else {
    await getPool().query(`
      DELETE FROM metalab_instalab_likes likes USING metalab_instalab_posts post
      WHERE likes.post_id = post.id AND likes.post_id = $1 AND likes.student_id = $2
        AND EXISTS (SELECT 1 FROM metalab_businesses WHERE student_id = $2)
        AND post.class_id = (SELECT assignment.class_id FROM metalab_student_class_assignments assignment WHERE assignment.student_id = $2)
    `, [postId, studentId]);
  }
  const result = await getPool().query<{ like_count: number | string; liked_by_me: boolean }>(`
    SELECT COUNT(likes.student_id) AS like_count, BOOL_OR(likes.student_id = $2) AS liked_by_me
    FROM metalab_instalab_posts post
    JOIN metalab_student_class_assignments assignment ON assignment.student_id = $2 AND assignment.class_id = post.class_id
    JOIN metalab_businesses business ON business.student_id = $2
    LEFT JOIN metalab_instalab_likes likes ON likes.post_id = post.id
    WHERE post.id = $1 GROUP BY post.id
  `, [postId, studentId]);
  return result.rows[0] ? { likeCount: Number(result.rows[0].like_count), likedByMe: result.rows[0].liked_by_me === true } : null;
}

export async function saveMissionAttempt(input: {
  studentId: string;
  missionId: string;
  status: MissionStatus;
  answer: unknown;
  evidence: Record<string, string | number | boolean>;
  feedback: string;
}) {
  const result = await getPool().query<MissionRow>(`
    INSERT INTO metalab_mission_progress
      (id, student_id, mission_id, status, answer, evidence, feedback, completed_at)
    VALUES ($1, $2, $3, $4::text, $5::JSONB, $6::JSONB, $7,
      CASE WHEN $4 = 'completed' THEN NOW() ELSE NULL END)
    ON CONFLICT (student_id, mission_id) DO UPDATE SET

        WHEN metalab_mission_progress.status = 'completed' THEN 'completed'

      END,

        WHEN metalab_mission_progress.status = 'completed' AND EXCLUDED.status <> 'completed'

          THEN metalab_mission_progress.completed_at

        ELSE NULL
      END
    RETURNING mission_id, status, answer, evidence, feedback, attempt_count,
              updated_at, completed_at
  `, [randomUUID(), input.studentId, input.missionId, input.status,
    JSON.stringify(input.answer), JSON.stringify(input.evidence), input.feedback]);

# Histórico diário de atividades dos alunos

O painel do professor exibe uma linha do tempo por aluno na aba **Alunos**. Ela permite consultar um dia, intervalo ou histórico completo, avançar/voltar dias, localizar dias que já têm eventos, paginar registros e imprimir relatórios em PDF pelo diálogo de impressão do navegador.

## Arquitetura

A migração `011_student_activity_history.sql` cria `metalab_student_activity_events`, um registro append-only com UUIDs/referências dos dados relacionados, instante `TIMESTAMPTZ`, fuso de apresentação `America/Sao_Paulo`, tipo de ator, turma, entidade e payload JSONB. Índices atendem consulta por aluno, ator e turma. Gatilhos PostgreSQL gravam os eventos dentro da mesma transação da gravação original. A rota de leitura exige sessão válida do professor, valida o UUID, limita páginas a 100 eventos e usa parâmetros SQL.

O histórico inclui cadastro, arquivamento/restauração, empresa, progresso/tentativas de missões, tarefas e entregas, publicações/edições/exclusões, curtidas e comentários do InstaLab, fotos recebidas/atualizadas/removidas, associação a turmas e início/encerramento explícito de sessões. Curtidas/comentários aparecem para o autor e para o aluno dono da publicação. Respostas, legendas e comentários ficam em snapshots; bytes de fotos e anexos, hashes/tokens de sessão, CPF e senhas não entram no histórico.

## Limites dos dados anteriores

A migração recupera apenas datas e fatos ainda presentes nas tabelas: cadastro, envio/recebimento de arquivos, respostas da Missão da Turma, publicações existentes, comentários/curtidas atuais, associação de turma registrada e sessões criadas/revogadas. Datas de foto representam o recebimento pelo servidor; a captura informada pelo celular não é considerada confiável.

Para registros antigos do InstaLab, a data de criação é preservada, mas a legenda original pode ter sido editada e não existe versão histórica no banco; o evento antigo informa essa limitação. Uma empresa e cada progresso de missão recebem um marco de estado atual na instalação da migração; versões anteriores e respostas de tentativas sobrescritas não podem ser reconstruídas. O sistema não atribui esse estado a uma data anterior nem cria eventos fictícios. Transferências de turma anteriores à migração também não podem ser recuperadas; somente o vínculo corrente e seu `assigned_at` permanecem conhecidos.

Depois que a migração estiver ativa, cada nova alteração de empresa, tentativa de missão e interação é preservada com os estados anterior/posterior disponíveis. Entregas e envios já são registros próprios e passam a gerar eventos automaticamente.

## Implantação segura

1. Faça backup/snapshot do banco que será migrado.
2. Confira, sem revelar o valor, que `DATABASE_URL` aponta para o banco esperado e que suas migrations anteriores estão aplicadas.
3. Em um banco de teste descartável, aplique a migration e valide criação de empresa, atualização, tentativa repetida, comentário/curtida, envio de tarefa, foto, troca de turma e consulta como professor.
4. Só então aplique ao banco real com `npm run db:migrate`. O comando aplica migrations pendentes em ordem e registra cada uma em `metalab_schema_migrations`.
5. Implante a aplicação com a mesma migration 011. Antes da migration, a nova rota retorna indisponibilidade; depois dela, a aba de alunos poderá consultar os eventos.

Não rode `npm run db:migrate` enquanto `DATABASE_URL` não tiver sido conferida. Nesta implementação a migration 011 foi criada localmente e **não foi executada** contra Railway nem outro banco real.

## PDFs

Os botões **PDF do dia**, **PDF do período** e **PDF do histórico completo** carregam todas as páginas necessárias e abrem o diálogo do navegador; selecione **Salvar como PDF**. Para evitar documentos excessivos, a impressão completa limita-se a 5.000 eventos e solicita período menor se esse limite for excedido.
